    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // 1. Toggle Chat History Panel
            const chatInputArea = document.getElementById('chatInputArea');
            const chatHistory = document.getElementById('chatHistory');
            let isChatOpen = true;

            chatInputArea.addEventListener('click', (e) => {
                // Prevent trigger if clicking on the actual textarea or buttons inside
                if(e.target.tagName !== 'DIV' && e.target !== chatInputArea) return;

                isChatOpen = !isChatOpen;
                if (isChatOpen) {
                    chatHistory.classList.remove('opacity-0', 'translate-y-4', 'pointer-events-none');
                    chatHistory.classList.add('opacity-100', 'translate-y-0');
                    // Ensure space is taken up
                    chatHistory.style.height = 'auto'; 
                    chatHistory.style.marginBottom = '1rem';
                } else {
                    chatHistory.classList.remove('opacity-100', 'translate-y-0');
                    chatHistory.classList.add('opacity-0', 'translate-y-4', 'pointer-events-none');
                    // Collapse space smoothly
                    setTimeout(() => {
                        if(!isChatOpen) { // double check state hasn't changed back
                            chatHistory.style.height = '0px';
                            chatHistory.style.marginBottom = '0px';
                            chatHistory.style.overflow = 'hidden';
                        }
                    }, 300); // Wait for transition to finish
                }
            });

            // Set initial state based on default HTML classes
            chatHistory.style.transition = 'all 0.3s cubic-bezier(0.4, 0, 0.2, 1)';
            chatHistory.style.marginBottom = '1rem';


            // 2. Toggle Right Sidebar
            const toggleRightPanelBtn = document.getElementById('toggleRightPanelBtn');
            const rightPanel = document.getElementById('rightPanel');
            let isRightPanelOpen = true;

            toggleRightPanelBtn.addEventListener('click', () => {
                isRightPanelOpen = !isRightPanelOpen;
                if (isRightPanelOpen) {
                    rightPanel.classList.remove('translate-x-full', 'opacity-0', 'pointer-events-none');
                    rightPanel.classList.add('translate-x-0', 'opacity-100');
                    toggleRightPanelBtn.classList.add('bg-gray-100');
                } else {
                    rightPanel.classList.remove('translate-x-0', 'opacity-100');
                    rightPanel.classList.add('translate-x-full', 'opacity-0', 'pointer-events-none');
                    toggleRightPanelBtn.classList.remove('bg-gray-100');
                }
            });


            // 3. Iframe Wrapper Scaling Logic
            // The iframe wrapper needs to look like a desktop screen floating in the center.
            const container = document.getElementById('iframeContainer');
            const wrapper = document.getElementById('iframeWrapper');
            
            let currentScale = 1;
            let panX = 0;
            let panY = 0;

            function updateTransform() {
                wrapper.style.transform = `translate(${panX}px, ${panY}px) scale(${currentScale})`;
            }
            
            function scaleWrapper() {
                // Get available space in container
                const containerWidth = container.clientWidth;
                const containerHeight = container.clientHeight;
                
                // Original simulated size
                const baseWidth = 1200;
                const baseHeight = 800;

                // Add padding so it doesn't touch edges
                const padding = 120; // Accounts for top bar, bottom bar, and sidebars generally
                
                const availableWidth = containerWidth - padding;
                const availableHeight = containerHeight - padding;

                // Calculate scale needed to fit width and height
                const scaleX = availableWidth / baseWidth;
                const scaleY = availableHeight / baseHeight;

                currentScale = Math.min(scaleX, scaleY);
                
                // Hardcap the scale so it always looks like a preview, never full screen
                currentScale = Math.min(currentScale, 0.75); 
                panX = 0;
                panY = 0;

                // Apply transform
                updateTransform();
            }

            // Initial scale
            scaleWrapper();

            // Rescale on window resize
            window.addEventListener('resize', scaleWrapper);

            // Canvas Zoom and Pan implementation
            container.addEventListener('wheel', (e) => {
                e.preventDefault(); // Prevent page scrolling

                // Check for pinch-to-zoom (trackpads often trigger ctrlKey) or explicit Ctrl+Scroll
                if (e.ctrlKey || e.metaKey) {
                    const zoomSensitivity = 0.005;
                    const delta = -e.deltaY * zoomSensitivity;
                    currentScale = Math.max(0.1, Math.min(currentScale + delta, 5));
                } else {
                    // Standard scroll behaves as a pan
                    panX -= e.deltaX;
                    panY -= e.deltaY;
                }
                updateTransform();
            }, { passive: false });

            // Allow mouse-drag to pan
            let isDraggingCanvas = false;
            let startDragX = 0;
            let startDragY = 0;

            container.addEventListener('mousedown', (e) => {
                // Drag if middle mouse clicked, OR if clicking directly on the empty background container
                if (e.button === 1 || e.target === container) {
                    isDraggingCanvas = true;
                    startDragX = e.clientX - panX;
                    startDragY = e.clientY - panY;
                    container.style.cursor = 'grabbing';
                    e.preventDefault();
                }
            });

            window.addEventListener('mousemove', (e) => {
                if (!isDraggingCanvas) return;
                panX = e.clientX - startDragX;
                panY = e.clientY - startDragY;
                updateTransform();
            });

            window.addEventListener('mouseup', () => {
                isDraggingCanvas = false;
                container.style.cursor = 'default';
            });
        });
    </script>
